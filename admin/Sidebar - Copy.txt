import React from 'react';
import { LayoutDashboard, Layers, LogOut, Hexagon } from 'lucide-react';

interface SidebarProps {
    activeTab: string;
    setActiveTab: (tab: string) => void;
    isOpen: boolean;
    setIsOpen: (isOpen: boolean) => void;
    onLogout: () => void;
}

const Sidebar: React.FC<SidebarProps> = ({
                                             activeTab,
                                             setActiveTab,
                                             isOpen, // Added missing prop
                                             setIsOpen,
                                             onLogout
                                         }) => {
    const navItems = [
        { id: 'dashboard', label: 'Dashboard', icon: LayoutDashboard },
        { id: 'edit-questions', label: 'Repository', icon: Layers },
    ];

    return (
        <aside className={`
            h-full w-64 bg-white/60 dark:bg-[#202123]/60 backdrop-blur-2xl 
             border border-[#1c1c1e]/5 dark:border-[#f5f5f7]/10 
            shadow-[0_8px_32px_rgba(0,0,0,0.04)] flex flex-col relative overflow-hidden
            transition-transform duration-300 ease-in-out
            ${isOpen ? 'translate-x-0' : '-translate-x-full md:translate-x-0'}
        `}>

            {/* Branding */}
            <div className="pt-5 pb-5 px-5 flex items-center gap-3">
                <div className="p-2 bg-[#1c1c1e] dark:bg-[#f5f5f7] rounded-xl shadow-sm">
                    <Hexagon size={20} strokeWidth={2} className="text-white dark:text-[#1c1c1e]" />
                </div>
                <span className="text-lg font-semibold tracking-wide text-[#1c1c1e] dark:text-[#f5f5f7]">Admin Panel</span>
            </div>

            {/* Navigation */}
            <nav className="flex-1 px-4 space-y-1.5">
                {navItems.map((item) => {
                    const isActive = activeTab === item.id;
                    const Icon = item.icon;

                    return (
                        <button
                            key={item.id}
                            onClick={() => {
                                setActiveTab(item.id);
                                setIsOpen(false); // Closes sidebar on mobile after click
                            }}
                            className={`
                                w-full flex items-center gap-4 px-5 py-3.5 rounded-2xl text-sm transition-all duration-200 relative group
                                ${isActive
                                ? 'text-[#1c1c1e] dark:text-[#f5f5f7] font-medium bg-[#1c1c1e]/5 dark:bg-[#f5f5f7]/10 shadow-sm'
                                : 'text-[#86868b] hover:text-[#1c1c1e] dark:hover:text-[#f5f5f7] hover:bg-[#1c1c1e]/[0.02] dark:hover:bg-[#f5f5f7]/[0.02]'
                            }
                            `}
                        >
                            {/* Left indicator bar for active state */}
                            {isActive && (
                                <div className="absolute left-0 top-1/2 -translate-y-1/2 w-1 h-1/2 bg-[#1c1c1e] dark:bg-[#f5f5f7] rounded-r-full" />
                            )}
                            <Icon size={18} strokeWidth={isActive ? 2.5 : 1.5} className="relative z-10" />
                            <span className="relative z-10">{item.label}</span>
                        </button>
                    );
                })}
            </nav>

            {/* User Profile */}
            <div className="p-4 mt-auto">
                <div className="bg-[#1c1c1e]/[0.03] dark:bg-[#f5f5f7]/[0.05] rounded-[1.5rem] p-3 pl-4 flex items-center justify-between transition-colors hover:bg-[#1c1c1e]/[0.06] dark:hover:bg-[#f5f5f7]/[0.1]">
                    <div className="flex items-center gap-3">
                        <div className="w-9 h-9 rounded-full bg-gradient-to-tr from-[#1c1c1e] to-[#4a4a4c] dark:from-[#f5f5f7] dark:to-[#d2d2d7] text-white dark:text-[#1c1c1e] flex items-center justify-center text-xs font-semibold shadow-md">
                            AC
                        </div>
                        <div className="flex flex-col text-left">
                            <span className="text-sm font-medium text-[#1c1c1e] dark:text-[#f5f5f7]">Alex C.</span>
                            <span className="text-[10px] text-[#86868b] uppercase tracking-widest font-medium">Architect</span>
                        </div>
                    </div>
                    <button
                        onClick={onLogout}
                        className="p-2 text-[#86868b] hover:text-red-500 hover:bg-red-50 dark:hover:bg-red-500/10 rounded-xl transition-all"
                        aria-label="Logout"
                    >
                        <LogOut size={16} strokeWidth={2} />
                    </button>
                </div>
            </div>
        </aside>
    );
};

export default Sidebar;
